def log_info(msg):    print(f"{CYAN}[INFO]{RESET} {msg}")
def log_success(msg): print(f"{GREEN}[OK]{RESET} {msg}")
def log_warning(msg): print(f"{YELLOW}[WARN]{RESET} {msg}")

class TempMailService:
    def __init__(self, 16029|h2SdEQm0tUYsyRojqKYVbNMdNln9loGyUGLF1wph8efb7b0d, base_url):
        self.api_token = 16029|h2SdEQm0tUYsyRojqKYVbNMdNln9loGyUGLF1wph8efb7b0d
        self.base_url = base_url.rstrip("/")
        self.current_email = None
        self.seen_ids = set()

    def test_api(self):

            ("GET", f"{self.base_url}/api/email/new"),
            ("GET", f"{self.base_url}/api/v1/email/new"),

            ("GET", f"{self.base_url}/api/inbox/new"),
            ("GET", f"{self.base_url}/api/domains"),

                if method == "GET":
                    resp = requests.get(url, headers=self.headers, timeout=10)

        return None

    def _extract_email_from_response(self, data):
        if isinstance(data, dict):
            for path in [["email"], ["address"], ["data","email"], ["data","address"],
                         ["result","email"], ["result","address"], ["email_address"]]:
                current = data
                try:
                    for key in path:
                        current = current[key]
                    if current and "@" in str(current):
                        return str(current)
                except (KeyError, TypeError):
                    continue
        if isinstance(data, str) and "@" in data:
            return data
        return None

    def get_messages(self):
        if not self.current_email:
            return []
        endpoints = [
            f"{self.base_url}/api/email/messages",
            f"{self.base_url}/api/messages",
            f"{self.base_url}/api/inbox",
            f"{self.base_url}/api/emails",
            f"{self.base_url}/api/email/inbox",
        ]
        for url in endpoints:
            try:
                resp = requests.get(url, headers=self.headers,
                                    params={"email": self.current_email}, timeout=15)
                if resp.status_code == 200:
                    msgs = self._extract_messages(resp.json())
                    if msgs is not None:
                        return msgs
                resp = requests.get(f"{url}/{self.current_email}",
                                    headers=self.headers, timeout=15)
                if resp.status_code == 200:
                    msgs = self._extract_messages(resp.json())
                    if msgs is not None:
                        return msgs
            except Exception:
                continue
        return []

    def _extract_messages(self, data):
        if isinstance(data, list):
            return data
        if isinstance(data, dict):
            for key in ["messages", "data", "emails", "inbox", "result", "items"]:
                if key in data:
                    val = data[key]
                    if isinstance(val, list):
                        return val
                    if isinstance(val, dict):
                        for sub in ["messages", "emails", "items"]:
                            if sub in val and isinstance(val[sub], list):
                                return val[sub]
        return None

    def wait_for_code(self, timeout=240):
        log_info(f"Waiting for code ({timeout}s)...")
        start = time.time()
        dots = 0
        while time.time() - start < timeout:
            for msg in self.get_messages():
                mid = msg.get("id") or msg.get("_id") or msg.get("uid")
                if mid and mid in self.seen_ids:
                    continue
                if mid:
                    self.seen_ids.add(mid)
                body = self._get_body(msg)
                code = self._extract_code(body)
                if code:
                    print()
                    log_success(f"Code: {code}")
                    return code
            dots += 1
            print(".", end="", flush=True)
            if dots % 12 == 0:
                print(f" [{int(time.time()-start)}s]", end="", flush=True)
            time.sleep(CONFIG["poll_interval"])
        print()
        log_warning("Timeout")
        return None

    def _get_body(self, msg):
        if isinstance(msg, str):
            return msg
        parts = []
        for key in ["body", "text", "html", "content", "message", "snippet", "subject"]:
            if key in msg and msg[key]:
                parts.append(str(msg[key]))
        if "data" in msg and isinstance(msg["data"], dict):
            for key in ["body", "text", "html", "content"]:
                if key in msg["data"] and msg["data"][key]:
                    parts.append(str(msg["data"][key]))
        return " ".join(parts)

    @staticmethod
    def _extract_code(text):
        if not text:
            return None
        for pattern in [
            r"\b(\d{6})\b",
            r"\b(\d{5})\b",
            r"code[\s:]+(\d{4,6})",
            r"(\d{6})\s+is your",
            r">\s*(\d{6})\s*<",
            r"FB-(\d{6})",
        ]:
            match = re.search(pattern, text, re.IGNORECASE | re.MULTILINE)
            if match:
                return match.group(1)
        return None


def save_account(email, password, filename):
    with open(filename, "a", encoding="utf-8") as f:
        f.write(f"{email}:{password}\n")
    log_success(f"Saved to {filename}")


def download_file(filename):
    try:
        from google.colab import files
        files.download(filename)
        log_success("File downloaded")
    except Exception as e:
        log_warning(f"Download skipped: {e}")
